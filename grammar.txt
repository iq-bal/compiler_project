program           -> declarations functions main
main              -> MOKSHA LPAREN RPAREN LBRACE statements RBRACE  
declarations      -> declarations declaration SEMICOLON | /* empty */
declaration       -> type IDENTIFIER | type IDENTIFIER ASSIGN value expression
functions         -> functions function_declaration | /* empty */ 
function_declaration -> type IDENTIFIER LPAREN parameters RPAREN LBRACE statements RBRACE
parameters        -> parameters_list | /* empty */
parameters_list   -> parameter | parameters_list COMMA parameter
parameter        -> type IDENTIFIER

statements        -> statements statement | /* empty */
statement         -> declaration SEMICOLON 
                    | assignment SEMICOLON 
                    | conditional 
                    | loop 
                    | function_call SEMICOLON 
                    | return_statement SEMICOLON 
                    | input_output SEMICOLON 
                    | for_loop 
                    | break SEMICOLON 
                    | continue SEMICOLON


for_loop         -> YATRA LPAREN for_initialization SEMICOLON for_condition SEMICOLON for_increment RPAREN LBRACE statements RBRACE

for_initialization  -> /* empty */ | assignment
for_condition       -> /* empty */ | expression
for_increment       -> /* empty */ | assignment




assignment        -> IDENTIFIER ASSIGN expression   
conditional       -> DHARMA LPAREN expression RPAREN LBRACE statements RBRACE
                    | DHARMA LPAREN expression RPAREN LBRACE statements RBRACE ADHARMA LBRACE statements RBRACE    
loop              -> TAPAS LPAREN expression RPAREN LBRACE statements RBRACE

expression        -> term 
                    | expression YOGA term 
                    | expression VIYOGA term 
                    | expression VIBHAGA term 
                    | expression OR term
                    | expression LT term
                    | expression GT term
                    | expression LE term
                    | expression GE term
                    | expression EQ term
                    | expression NE term


                    

term              -> factor | term BAL factor | term VIBHAJAN factor | term VIBHAGA factor

factor            -> LPAREN expression RPAREN | IDENTIFIER | literal | function_call

literal           -> INTEGER_LITERAL | STRING_LITERAL

function_call     -> IDENTIFIER LPAREN arguments RPAREN

arguments         -> arguments_list | /* empty */

arguments_list    -> expression | arguments_list COMMA expression

input_output      -> INPUT LPAREN IDENTIFIER RPAREN | OUTPUT LPAREN expression RPAREN

type              -> ARTHA | JALA | SATYA | AKSHARA | VANI
